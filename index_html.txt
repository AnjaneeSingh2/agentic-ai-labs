<script>
const log = document.getElementById('log');
const input = document.getElementById('input');
const sendBtn = document.getElementById('send');
const statusEl = document.getElementById('status');
const statusText = document.getElementById('status-text');

function addRow(who, text, citations) {
  const row = document.createElement('div');
  row.className = 'row ' + who;
  const whoEl = document.createElement('div');
  whoEl.className = 'who';
  whoEl.textContent = who === 'you' ? 'you' : (who === 'err' ? 'error' : 'agent');
  const msgEl = document.createElement('div');
  msgEl.className = 'msg';
  msgEl.textContent = text;
  row.appendChild(whoEl);
  row.appendChild(msgEl);

  if (citations && citations.length) {
    const cites = document.createElement('div');
    cites.className = 'cites';
    citations.forEach((c, i) => {
      const d = document.createElement('div');
      if (c.url) {
        d.innerHTML = `[${i+1}] <a href="${c.url}" target="_blank" rel="noopener">${escapeHtml(c.title)}</a>`;
      } else {
        d.textContent = `[${i+1}] ${c.title}`;
      }
      cites.appendChild(d);
    });
    msgEl.appendChild(cites);
  }

  log.appendChild(row);
  log.scrollTop = log.scrollHeight;
}

function escapeHtml(s) {
  const d = document.createElement('div');
  d.textContent = s;
  return d.innerHTML;
}

async function checkHealth() {
  try {
    const res = await fetch('/api/health');
    const data = await res.json();
    if (data.ok) {
      statusEl.className = 'status ok';
      statusText.textContent = 'ready';
    } else {
      statusEl.className = 'status err';
      statusText.textContent = data.error || 'not configured';
    }
  } catch (e) {
    statusEl.className = 'status err';
    statusText.textContent = 'unreachable';
  }
}

async function send() {
  const text = input.value.trim();
  if (!text) return;
  input.value = '';
  sendBtn.disabled = true;
  addRow('you', text);

  const thinking = document.createElement('div');
  thinking.className = 'row agent';
  thinking.innerHTML = '<div class="who">agent</div><div class="msg">…</div>';
  log.appendChild(thinking);
  log.scrollTop = log.scrollHeight;

  try {
    const res = await fetch('/api/chat', {
      method: 'POST',
      headers: {'Content-Type': 'application/json'},
      body: JSON.stringify({message: text})
    });
    const data = await res.json();
    thinking.remove();
    if (!res.ok) {
      addRow('err', data.error || 'Request failed.');
    } else {
      addRow('agent', data.reply, data.citations);
    }
  } catch (e) {
    thinking.remove();
    addRow('err', 'Network error: ' + e.message);
  } finally {
    sendBtn.disabled = false;
    input.focus();
  }
}

sendBtn.addEventListener('click', send);
input.addEventListener('keydown', (e) => {
  if (e.key === 'Enter') send();
});

checkHealth();
